# Dependências de runtime do SynapseShop.
# Aula 4: Django REST Framework integrado ao ambiente conteinerizado
# (PostgreSQL via psycopg, WSGI via gunicorn).
Django>=5.2,<6.0
djangorestframework>=3.16,<4.0
dj-database-url>=2.3,<3.0
psycopg[binary]>=3.2,<4.0
gunicorn>=23.0,<24.0
# Aula 7: autenticação JWT sobre o auth.User do Django (SimpleJWT) e
# filtros declarativos do DRF (django-filter) para listas paginadas.
djangorestframework-simplejwt>=5.5,<6.0
django-filter>=25,<26
# Aula 8: cliente Redis (redis-py). É a única dependência nova — ele é
# consumido pelo backend de cache NATIVO do Django
# (django.core.cache.backends.redis.RedisCache), que já fala SETEX/INCR.
redis>=5.0,<7.0
# Aula 9: cliente AMQP 0-9-1 (RabbitMQ). Escolhido em vez de Celery/kombu
# porque a spec pede que a política de reentrega e o roteamento para a DLQ
# sejam visíveis no código do projeto, e não atrás de decorators.
pika>=1.3,<2.0
# Aula 10: cliente Kafka (`confluent-kafka`, sobre o librdkafka em C).
# Escolhido em vez de `kafka-python` porque é o único cliente que expõe
# o que a spec pede: delivery report com `acks=all`, produtor
# idempotente, consumer group com commit manual de offset e a API de
# administração (criar tópico com partições/retenção explícitas) — sem ela a
# topologia teria de ser criada por script externo ao projeto.
# O pin fica em >=2.15,<3.0 porque `enable.idempotence` e a semântica de
# commit por mensagem (`commit(message=...)`) que o consumidor usa são
# garantias da série 2.x; wheels existentes para cp312 (Dockerfile) e cp314
# (venv local), então a instalação não cai no build do librdkafka.
confluent-kafka>=2.15,<3.0
